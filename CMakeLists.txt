cmake_minimum_required(VERSION 3.20)
project(bmatrix VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BMATRIX_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
find_package(PNG REQUIRED)
find_package(Curses REQUIRED)
add_library(bmatrix_options INTERFACE)
target_compile_options(bmatrix_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(BMATRIX_SANITIZERS)
  target_compile_options(bmatrix_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bmatrix_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(bmatrix_core STATIC src/core.cpp src/image.cpp)
target_include_directories(bmatrix_core PUBLIC include)
target_link_libraries(bmatrix_core PUBLIC bmatrix_options PRIVATE PNG::PNG)
add_executable(bmatrix src/main.cpp src/terminal.cpp)
target_include_directories(bmatrix PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(bmatrix PRIVATE bmatrix_core bmatrix_options ${CURSES_LIBRARIES})
# Some distributions keep terminfo symbols outside libncurses.
find_library(BMATRIX_TERMINFO_LIBRARY NAMES tinfo)
if(BMATRIX_TERMINFO_LIBRARY)
  target_link_libraries(bmatrix PRIVATE ${BMATRIX_TERMINFO_LIBRARY})
endif()
target_compile_definitions(bmatrix PRIVATE BMATRIX_VERSION="${PROJECT_VERSION}")
install(TARGETS bmatrix RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES man/bmatrix.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(bmatrix_core_test tests/core_test.cpp)
  target_link_libraries(bmatrix_core_test PRIVATE bmatrix_core bmatrix_options PNG::PNG)
  add_test(NAME core COMMAND bmatrix_core_test)
  add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_test.py $<TARGET_FILE:bmatrix>)
  set_tests_properties(terminal PROPERTIES TIMEOUT 45)
endif()
